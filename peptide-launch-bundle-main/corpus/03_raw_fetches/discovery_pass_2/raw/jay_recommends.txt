  var TRACKED_PARAMS = [
    'nk_source',
    'nk_medium',
    'nk_campaign',
    'nk_content',
    'utm_source',
    'utm_medium',
    'utm_campaign',
    'utm_content'

  var COOKIE_NAME = '_nk_attribution';
  var COOKIE_DOMAIN = '.jaycampbell.com';
  var COOKIE_PATH = '/';
  var COOKIE_EXPIRY_DAYS = 3;

   * only the attribution params that are present and non-empty.

  function getAttributionParams() {

    var attribution = {};

    for (var i = 0; i < TRACKED_PARAMS.length; i++) {
