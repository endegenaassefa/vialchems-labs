(function () {

  const PARAM = "a_aid";
  const STORAGE_KEY = "rac_affiliate";

  function getAffiliateFromHash() {
    if (window.location.hash.includes(PARAM + "=")) {
      const hashParams = new URLSearchParams(window.location.hash.substring(1));
      return hashParams.get(PARAM);
    }
    return null;
  }

  function getAffiliateFromQuery() {
    const url = new URL(window.location.href);
    return url.searchParams.get(PARAM);
  }

  function saveAffiliate(aid) {
    if (!aid) return;
    localStorage.setItem(STORAGE_KEY, aid);
    document.cookie = STORAGE_KEY + "=" + aid + "; path=/; max-age=2592000";
  }

  function loadAffiliate() {
    let aid = localStorage.getItem(STORAGE_KEY);
    if (aid) return aid;
    const match = document.cookie.match(new RegExp(STORAGE_KEY + "=([^;]+)"));
    return match ? match[1] : null;
  }

  function redirectWithParam(aid) {
    if (!aid) return;

    const url = new URL(window.location.href);

    if (!url.searchParams.get(PARAM)) {
      url.searchParams.set(PARAM, aid);
      url.hash = "";   // remove hash
      window.location.replace(url.toString());
    }
  }

  // STEP 1 — capture from hash
  const hashAid = getAffiliateFromHash();
  if (hashAid) {
    saveAffiliate(hashAid);
    redirectWithParam(hashAid);
    return;
  }

  // STEP 2 — capture from query
  const queryAid = getAffiliateFromQuery();
  if (queryAid) {
    saveAffiliate(queryAid);
  }

  // STEP 3 — restore if missing
  const storedAid = loadAffiliate();
  if (storedAid) {
    redirectWithParam(storedAid);
  }

})();
