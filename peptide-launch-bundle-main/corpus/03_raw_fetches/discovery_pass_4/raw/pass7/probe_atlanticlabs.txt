<script>
    function testimonialSlider(totalTestimonials, visibleCount) {
        return {
            currentSlide: 0,
            totalTestimonials: totalTestimonials,
            visibleCount: visibleCount,
            isMobile: !window.matchMedia('(min-width: 768px)').matches,
            get itemsPerSlide() {
                return this.isMobile ? 1 : this.visibleCount;
            },
            get totalPages() {
                return Math.ceil(this.totalTestimonials / this.itemsPerSlide);
            },
            get slideWidth() {
                return this.isMobile ? 100 : (100 / this.visibleCount);
            },
            get canGoNext() {
                return true; 
            },
            nextSlide() {
                const maxSlide = this.totalPages - 1;
                this.currentSlide = this.currentSlide >= maxSlide ? 0 : this.currentSlide + 1;
            },
            previousSlide() {
                const maxSlide = this.totalPages - 1;
                this.currentSlide = this.currentSlide <= 0 ? maxSlide : this.currentSlide - 1;
            },
            goToPage(page) {
                this.currentSlide = page;
            },
            init() {
                
                const mq = window.matchMedia('(min-width: 768px)');
                this.isMobile = !mq.matches;
                mq.addEventListener('change', (e) => {
                    this.isMobile = !e.matches;
                    this.currentSlide = 0;
                });
            }
        }
    }
</script>
