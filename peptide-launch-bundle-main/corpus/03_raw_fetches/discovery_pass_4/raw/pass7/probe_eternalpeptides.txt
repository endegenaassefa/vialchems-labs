jQuery(function($){
    let isSubmitting = false;

    function hideMessages(){
        $("#success-msg, #error-msg").stop(true,true).hide();
        // error-msg টেক্সট আগেরটা রিস্টোর (যদি কখনো পরিবর্তন করা হয়)
        $("#error-msg").text("The email has already been subscribed.");
    }

    function showSuccess(){
        hideMessages();
        $("#success-msg").fadeIn();
        $("#EMAIL").val('');
    }

    function showAlreadySubscribed(){
        hideMessages();
        $("#error-msg").fadeIn();
    }

    function responseLooksAlreadySubscribed(text){
        if(!text) return false;
        const t = String(text).toLowerCase();
        if (t.includes("already been subscribed") || t.includes("already subscribed")) return true;
        if (t.includes("already") && t.includes("subscrib")) return true;
        if (t.includes("exists") && t.includes("subscrib")) return true;
        return false;
    }

    $("#acelle-form").validate({
        rules:{
            EMAIL:{
                required: true,
                email: true,
                remote: {
                    url: "https://mail.webrelife.com/lists/6920add0533bd/check-email",
                    type: "get",
                    data: { EMAIL: function(){ return $("#EMAIL").val(); } }
                }
            }
        },
        messages:{
            EMAIL:{ remote: "The email has already been subscribed." }
        },

        invalidHandler: function(event, validator){
            hideMessages();
            const hasRemoteError = validator.errorList.some(e => e.method === "remote");
            if(hasRemoteError){
                showAlreadySubscribed();
            }
        },

        submitHandler: function(form){
            if(isSubmitting) return;
            isSubmitting = true;

            hideMessages();
            const $btn = $("#submit-btn");
            $btn.prop("disabled", true);

            $.ajax({
                url: $(form).attr("action"),
                type: "POST",
                data: $(form).serialize(),

                success: function(res){
                    if(responseLooksAlreadySubscribed(res)){
                        showAlreadySubscribed();
                    } else {
                        showSuccess();
                    }
                },

                error: function(xhr){
                    const resText = xhr.responseText || "";
                    if(responseLooksAlreadySubscribed(resText)){
                        showAlreadySubscribed();
                    } else {
                        // Acelle/CORS/redirect case -> assume success
                        showSuccess();
                    }
                },

                complete: function(){
                    isSubmitting = false;
                    $btn.prop("disabled", false);
                }
            });
        }
    });
});
